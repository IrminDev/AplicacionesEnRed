apt-get install -y vsftpd

touch vsftpd.conf &&
echo "listen=YES
anonymous_enable=NO
local_enable=YES
write_enable=YES
chroot_local_user=YES
allow_writeable_chroot=YES
pasv_enable=YES
pasv_min_port=30000

mv vsftpd.conf /etc/

useradd -m ftpuser && echo "ftpuser:clave123" | chpasswd

mkdir -p /var/run/vsftpd/empty
